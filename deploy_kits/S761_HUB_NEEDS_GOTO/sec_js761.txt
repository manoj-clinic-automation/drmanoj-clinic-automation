    return one;
  }
  var head='<span class="dot '+cls+'"></span><span class="gn '+cls+'">'+L.length+'</span> <b>'+s761Esc(g.title)+'</b>'+
           (g.key==="info"?' <span class="gsum">— nothing to do</span>':' <span class="gsum">— '+s761Esc(g.summary||"")+'</span>')+
           (g.key==="days"?' <button onclick="event.preventDefault();event.stopPropagation();goSec(\'days\')">Open the days</button>':'');

  if(g.key==="arrived"){
    var nsize=0;
    (g.vendors||[]).forEach(function(v){
      body+='<div class="vrow" data-target="porders" onclick="s761Go(this)"><b>'+s761Esc(v.vendor)+' ('+v.n+')</b> ▸</div><ul class="sub">'+
        (v.items||[]).map(function(x){ if(x.size) nsize++; return '<li class="vitem">'+s761Esc(x.item)+(x.note?' <span class="vnote">— '+s761Esc(x.note)+'</span>':'')+'</li>'; }).join("")+'</ul>';

function s761Bad(list){ return (list||[]).some(function(x){return x.status!=="ok"}); }
function s761BankLine(j){
  function cnt(list,word){ var L=list||[], ok=L.filter(function(x){return x.status==="ok"}).length, bad=L.filter(function(x){return x.status==="bad"}).length, w=L.length-ok-bad;
    if(!L.length) return "no "+word+"s";
    return ok+" "+word+(ok===1?"":"s")+" ✓"+(w?(" · "+w+" waiting for a statement"):"")+(bad?(' · <b class="bad">'+bad+" not in the statement</b>"):""); }
  var all=!s761Bad(j.deposits)&&!s761Bad(j.transfers)&&((j.deposits||[]).length+(j.transfers||[]).length>0);
  var MON=["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"], ym=String(j.month||"");
  return '<div class="cb-line">'+cnt(j.deposits,"deposit")+' · '+cnt(j.transfers,"transfer")+(all?' <span class="ok">— all confirmed</span>':'')+
    (ym.length===7?(' · Pharmacy UPI '+MON[parseInt(ym.slice(5,7),10)-1]+' ₹'+esc(j.upi_total)):'')+'</div>';
}
