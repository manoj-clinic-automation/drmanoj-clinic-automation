def _verify_blocks(ym):
    """STEP 0 data: absent_worklist, each staff's open and corrected days in one
    date-ordered list, WITHOUT the staff the policy lists as dates-only (their
    punch dates are returned separately). Returns (blocks, feed_ok, donly)."""
    P, _err = _policy_module()
    try:
        names = P.dates_only_names(P.load_settings()) if P else set()
    except Exception:
        names = set()
    con = get_db()
    try:
        blocks, feed_ok = absent_worklist(con, ym)
        staff_rows = [(r["staff_id"], r["name"]) for r in
                      con.execute("SELECT staff_id, name FROM staff WHERE active=1")]
    finally:
        con.close()
    blocks = [b for b in blocks if (b["name"] or "").strip().lower() not in names]
    for b in blocks:
        items = [dict(r, done=False) for r in b["rows"]] + [dict(r, done=True) for r in b["done"]]
        b["items"] = sorted(items, key=lambda r: r["d"])
    donly = []
    if names and feed_ok:
        idx = punch_month_index(ym) or {}
        for sid, nm in staff_rows:
            if (nm or "").strip().lower() in names:
                days = []
                for d in sorted(k for k, v in idx.items() if sid in v):
                    try:
                        dd = datetime.date.fromisoformat(d)
                    except ValueError:
                        continue
                    days.append({"day": dd.day, "sun": dd.weekday() == 6})
                donly.append({"name": nm, "days": days})
    return blocks, feed_ok, donly


@app.route(APP_PREFIX + "/salary/flow/verify")
@require("salary")
def salary_flow_verify():
    ym = _flow_ym()
    blocks, feed_ok, donly = _verify_blocks(ym)
    try:
        month = datetime.date(int(ym[:4]), int(ym[5:7]), 1).strftime("%B %Y")
    except Exception:
        month = ym
    return render_template_string(
        VERIFY_HTML, prefix=APP_PREFIX, ym=ym, month=month, blocks=blocks, feed_ok=feed_ok,
        donly=donly, today=_today(), nstaff=len(blocks),
        nopen=sum(len(b["rows"]) for b in blocks), ndone=sum(len(b["done"]) for b in blocks))
