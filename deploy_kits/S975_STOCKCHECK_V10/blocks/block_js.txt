function trk(T){return '<span class="trk">'+(T||[]).map(function(s){return '<span class="st '+esc(s[0])+'">'+esc(s[1])+'</span>';}).join('<span class="ar">→</span>')+'</span>';}

  +(r.tags||[]).map(tg).join('')+trk(r.track)+'<span class="sumfig">'+esc(r.cost_t)+'</span>'+acts(r)+'</summary><div class="inner">'+pth(r.path)+vlt(r.vlines)+led(r.key)+'</div></details>';}
function pdate(s){s=String(s||'');return /^\d\d-\d\d-\d{4}$/.test(s)?s.slice(6)+'-'+s.slice(3,5)+'-'+s.slice(0,2):'';}

  if(x.surplus&&D&&D.reports){var pt=pdate(D.reports.purchase_to);if(pt&&pt<x.day)x.tags.push(['blu','likely a delivery not yet in Marg (Marg\'s purchases reach '+D.reports.purchase_to.slice(0,5)+')']);}
  x.fig=(withDay?dm2(x.day)+' ':'')+x.time+(x.by?' · '+x.by:'')+' · '+x.fig;return drow(x);}
function b3fold(id,title,n,node){if(!node)return '';var t=node.querySelector('h2');if(t)t.remove();
