  function sharpS618(page){
    var sm = null, g = null, g6 = null, lap = null, mean = null, sd = null;
    try {
      var s = Math.min(1, 600 / Math.max(page.width, page.height));
      sm = CV.imread(page); g = new CV.Mat(); g6 = new CV.Mat(); lap = new CV.Mat(); mean = new CV.Mat(); sd = new CV.Mat();
      CV.cvtColor(sm, g, CV.COLOR_RGBA2GRAY);
      // shrunk by the library's own area average (the same on every phone; a canvas shrink aliases and reads print as sharpness)
      CV.resize(g, g6, new CV.Size(Math.max(8, Math.round(page.width * s)), Math.max(8, Math.round(page.height * s))), 0, 0, CV.INTER_AREA);
      CV.Laplacian(g6, lap, CV.CV_64F, 1, 1, 0, CV.BORDER_DEFAULT);                    // ksize 1 = the 3x3 kernel [0 1 0; 1 -4 1; 0 1 0]
      CV.meanStdDev(lap, mean, sd);
      return sd.data64F[0] * sd.data64F[0];
